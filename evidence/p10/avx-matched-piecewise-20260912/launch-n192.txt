# Bind the watchdog to the full solver process identity, not the time parent.
# Poll briefly until exactly one direct child appears; refuse ambiguity.
SOLVER_PID=
for _ in $(seq 1 100); do
    CHILDREN=$(pgrep -P "$TIME_PID" || true)
    if test "$(printf '%s\n' "$CHILDREN" | sed '/^$/d' | wc -l)" -eq 1; then
        SOLVER_PID=$CHILDREN
        break
    fi
    sleep 0.05
done
test -n "$SOLVER_PID"
PROC_STAT=$(cat "/proc/$SOLVER_PID/stat")
PROC_REST=${PROC_STAT##*) }
set -- $PROC_REST
test "$3" = "$TIME_PID"
PROCESS_GROUP=$3
STARTTIME=${20}
CMDLINE_SHA256=$(sha256sum "/proc/$SOLVER_PID/cmdline" | awk '{print $1}')
setsid "$WATCHDOG" "$SOLVER_PID" "$PROCESS_GROUP" "$STARTTIME" "$CMDLINE_SHA256" "$DEADLINE_EPOCH" "$LOG_DIR/watchdog.log" >"$LOG_DIR/watchdog.stdout" 2>"$LOG_DIR/watchdog.stderr" &
WATCHDOG_PID=$!
