# Before continuing, record fresh MemAvailable and filesystem availability,
# verify MemAvailable >= 113845656328 B and filesystem available >= 137438953472 B,
# verify the matched N192 solver has exited, and record a measured forecast whose

setsid /usr/bin/time -v -o "$LOG_DIR/time.txt" "$BIN" run "$OUTPUT_PARENT/state" >"$LOG_DIR/stdout" 2>"$LOG_DIR/stderr" &
TIME_PID=$!

# Bind the true v2 watchdog to the full solver child identity, not the time parent.
SOLVER_PID=
for _ in $(seq 1 100); do
    CHILDREN=$(pgrep -P "$TIME_PID" || true)
    if test "$(printf '%s\n' "$CHILDREN" | sed '/^$/d' | wc -l)" -eq 1; then
        SOLVER_PID=$CHILDREN
        break
    fi
    sleep 0.05
done
test -n "$SOLVER_PID"
PROC_STAT=$(cat "/proc/$SOLVER_PID/stat")
PROC_REST=${PROC_STAT##*) }
set -- $PROC_REST
test "$3" = "$TIME_PID"
PROCESS_GROUP=$3
STARTTIME=${20}
CMDLINE_SHA256=$(sha256sum "/proc/$SOLVER_PID/cmdline" | awk '{print $1}')
setsid "$WATCHDOG" "$SOLVER_PID" "$PROCESS_GROUP" "$STARTTIME" "$CMDLINE_SHA256" "$DEADLINE_EPOCH" "$LOG_DIR/watchdog.log" >"$LOG_DIR/watchdog.stdout" 2>"$LOG_DIR/watchdog.stderr" &
WATCHDOG_PID=$!

Record TIME_PID, SOLVER_PID, PROCESS_GROUP, STARTTIME, CMDLINE_SHA256,
